# Spell: Dream 001 - Mystery of Mental Domination
zz_valyria_magic_dream_magic.001 = {
	type = character_event
	window = zz_valyria_magic_character_event
	title = zz_valyria_magic_dream_magic.001.t
	desc = {
		random_valid = {
			desc = zz_valyria_magic_dream_magic.001.desc
			desc = zz_valyria_magic_dream_magic.001.desc_02
			desc = zz_valyria_magic_dream_magic.001.desc_03
		}
	}
	theme = university
	override_background = {
		reference = zz_dream_magic
	}
	lower_left_portrait = root
	lower_right_portrait = {
		character = scope:secondary_recipient
	}
	lower_center_portrait = scope:recipient
	immediate = {
		remove_variable = selected_flag
		clear_variable_list = possible_flags
		add_to_variable_list = { name = possible_flags target = flag:zz_domination_mind }
		add_to_variable_list = { name = possible_flags target = flag:zz_domination_soul }
		set_variable = { name = selected_flag value = flag:zz_domination_mind }
	}
	widgets = {
		widget = {
			is_shown = { is_ai = no }
			gui = "event_window_flag_selection"
			container = "custom_widgets_container"
		}
	}
	option = {
		name = zz_valyria_magic_dream_magic.001.opt.a
		hidden_effect = {
			switch = {
				trigger = var:selected_flag
				flag:zz_domination_mind = {
					save_scope_value_as = { name = option_01 value = yes }
				}
				flag:zz_domination_soul = {
					save_scope_value_as = { name = option_02 value = yes }
				}
			}
			zz_valyria_magic_cast_deferred_effect = {
				SKILL = dream_magic
				SPELL = 001
			}
		}
	}
	option = {
		name = zz_valyria_magic_dream_magic.001.opt.z
	}
}
