	theme = university
	override_background = {
		reference = zz_water_magic
	}
	lower_left_portrait = root

	option = {
		name = zz_valyria_magic_assassination.1001.a

			id = zz_valyria_magic_assassination.1002
			days = { 7 14 }
		}

			base = 30
		}
	}
	option = {
		name = zz_valyria_magic_assassination.1001.b

		zz_valyria_assassination_add_ward_effect = { WARD = 30 }

			id = zz_valyria_magic_assassination.1002
			days = { 7 14 }

		ai_chance = {
			base = 60
		}
	}
	option = {
		name = zz_valyria_magic_assassination.1001.c

			has_trait = zz_magister

# Shadow: the shade arrives, and most never know
zz_valyria_magic_assassination.2001 = {
	hidden = yes

	immediate = {
		if = {
			limit = {
				OR = {
					has_trait = zz_magister
					has_trait = paranoid
					intrigue >= 20
				}
			}
			trigger_event = zz_valyria_magic_assassination.2002
		}
		else = {
			trigger_event = {
				id = zz_valyria_magic_assassination.2003
				days = { 7 14 }
			}
		}
	}
}
