# Spell: Shadow 011 - Mystery of Umbral Strike (Regiment)
zz_valyria_magic_shadow_magic.011 = {
	type = character_event
	window = zz_valyria_magic_character_event
	title = zz_valyria_magic_shadow_magic.011.t
	desc = {
		random_valid = {
			desc = zz_valyria_magic_shadow_magic.011.desc
			desc = zz_valyria_magic_shadow_magic.011.desc_02
			desc = zz_valyria_magic_shadow_magic.011.desc_03
		}
	}
	theme = university 
	override_background = {
		reference = zz_shadow_magic
	}
	lower_left_portrait = root
	lower_right_portrait = {
		character = scope:secondary_recipient
	}
	lower_center_portrait = scope:recipient

	immediate = {
		clear_variable_list = possible_regiments
		remove_variable = selected_regiment
		scope:recipient = {
			every_maa_regiment = {
				limit = {
					maa_current_troops_count > 0
				}
				root = {
					add_to_variable_list = {
						name = possible_regiments
						target = prev
					}
					if = {
						limit = {
							NOT = { has_variable = selected_regiment }
						}
						set_variable = { name = selected_regiment value = prev }
					}
				}
			}
		}
	}
	widget = {
		is_shown = {
			root = {
				any_in_list = {
					variable = possible_regiments
					count > 0
				}
			}
		}
		gui = "event_window_maa_selection"
		container = "custom_widgets_container"
	}
	option = {
		name = zz_valyria_magic_shadow_magic.011.opt.a
		hidden_effect = {
			var:selected_regiment ?= { 
				save_scope_as = target
			}
			zz_valyria_magic_cast_deferred_effect = { 
				SKILL = shadow_magic
				SPELL = 011
			}	
		}
	}
	option = {
		name = zz_valyria_magic_shadow_magic.011.opt.z
	}
}
