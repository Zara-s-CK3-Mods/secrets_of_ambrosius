zz_valyria_shadow_magic_stat_gui = {
	if = {
		limit = {
			exists = this 
		}	
		value = "has_trait_xp(zz_magister|shadow_magic)" 
	}
	if = {
		limit = {
			has_variable = shadow_magic_overcap 
		}
		add = this.var:shadow_magic_overcap
	}
}
### Shadow Magic 
zz_valyria_magic_shadow_magic_base_value = { 
	scope:secondary_recipient = { 
		if = {
			limit = {
				exists = var:casttype
			}
			if = {
				limit = { 
					var:casttype ?= flag:potion 
				}
				value =  var:cast_potency
			}
			else_if = {
				limit = {
					var:casttype ?= flag:scroll
				}
				value =  var:cast_potency
			}
			else = { 
				value = "has_trait_xp(zz_magister|shadow_magic)" 
				if = {
					limit = {
						has_variable = shadow_magic_overcap 
					}
					add = this.var:shadow_magic_overcap
				}
			}
		}
		else = { 
			value = "has_trait_xp(zz_magister|shadow_magic)" 
			if = {
				limit = {
					has_variable = shadow_magic_overcap 
				}
				add = this.var:shadow_magic_overcap
			}
		}
		multiply = {
			value = 1
			if = {
				limit = {
					faith = { has_doctrine_parameter = zz_tenet_strength_shadow_magic }
				}
				add = 0.25
			}
			if = {
				limit = {
					has_personal_tenet_flag = zz_personal_strength_shadow_magic
				}
				add = 0.1
			}
		}
	}
}
