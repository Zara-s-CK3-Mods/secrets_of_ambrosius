zz_valyria_dream_magic_stat_gui = {
	if = {
		limit = {
			exists = this 
		}
		value = "has_trait_xp(zz_magister|dream_magic)" 
	}
	if = {
		limit = {
			has_variable = dream_magic_overcap 
		}
		add = this.var:dream_magic_overcap
	}
}
### Dream Magic!
zz_valyria_magic_dream_magic_base_value = { 
	scope:secondary_recipient = { 
		if = {
			limit = {
				exists = var:casttype
			}
			if = {
				limit = { 
					var:casttype ?= flag:potion 
				}
				value =  var:cast_potency
			}
			else_if = {
				limit = {
					var:casttype ?= flag:scroll
				}
				value =  var:cast_potency
			}
			else = { 
				value = "has_trait_xp(zz_magister|dream_magic)" 
				if = {
					limit = {
						has_variable = dream_magic_overcap 
					}
					add = this.var:dream_magic_overcap
				}
			}
		}
		else = { 
			value = "has_trait_xp(zz_magister|dream_magic)" 
			if = {
				limit = {
					has_variable = dream_magic_overcap 
				}
				add = this.var:dream_magic_overcap
			}
		}
		multiply = {
			value = 1
			if = {
				limit = {
					faith = { has_doctrine_parameter = zz_tenet_strength_dream_magic }
				}
				add = 0.25
			}
			if = {
				limit = {
					has_personal_tenet_flag = zz_personal_strength_dream_magic
				}
				add = 0.1
			}
		}
	}
}

		limit = { exists = root.var:zz_valyria_magic_event_window_counter_selection_counter_base }

		max = zz_valyria_magic_dream_magic_004_value_02
	}
	else = {
		value = zz_valyria_magic_dream_magic_004_value_02
	}
}
