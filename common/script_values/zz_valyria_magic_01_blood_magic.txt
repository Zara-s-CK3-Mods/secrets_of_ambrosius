zz_valyria_blood_magic_stat_gui = {
	if = {
		limit = {
			exists = this 
		}
		value = "has_trait_xp(zz_magister|blood_magic)" 
	}
	if = {
		limit = {
			has_variable = blood_magic_overcap 
		}
		add = this.var:blood_magic_overcap
	}
}
### Blood Magic
zz_valyria_magic_blood_magic_base_value = { 
	scope:secondary_recipient = { 
		if = {
			limit = {
				exists = var:casttype
			}
			if = {
				limit = { 
					var:casttype ?= flag:potion 
				}
				value = scope:actor.var:cast_potency
			}
			else_if = {
				limit = {
					var:casttype ?= flag:scroll
				}
				value =  scope:actor.var:cast_potency
			}
			else = { 
				value = "has_trait_xp(zz_magister|blood_magic)" 
				if = {
					limit = {
						has_variable = blood_magic_overcap 
					}
					add = this.var:blood_magic_overcap
				}
			}
		}
		else = { 
			value = "has_trait_xp(zz_magister|blood_magic)" 
			if = {
				limit = {
					has_variable = blood_magic_overcap 
				}
				add = this.var:blood_magic_overcap
			}
		}
		multiply = {
			value = 1
			if = {
				limit = {
					faith = { has_doctrine_parameter = zz_tenet_strength_blood_magic }
				}
				add = 0.25
			}
			if = {
				limit = {
					has_personal_tenet_flag = zz_personal_strength_blood_magic
				}
				add = 0.1
			}
		}
	}
}

	}
}
