zz_valyria_current_cost = {
	value = 0 
	if = {
		limit = {
			exists = this 
		}
		add = zz_valyria_current_item_cost
		add = zz_valyria_current_item_modifier
		divide = { # the crafter's skill lowers the cost, to half at most (Higher Mysteries 100)
			value = zz_valyria_current_crafter_stat
			max = 100
			divide = 50
			min = 1
		}
		# Mage's Power reagent: 100 mana takes 25% off the gold cost
		if = {
			limit = {
				var:zz_current_magic_reagent ?= flag:mages_power
			}
			multiply = 0.75
		}
		# Human Sacrifice reagent: a sacrifice takes half off the gold cost
		else_if = {
			limit = {
				var:zz_current_magic_reagent ?= flag:human_sacrifice
			}
			multiply = 0.5
		}
	}
}

# Potions tab: the chosen potion's cost, reduced by the crafter's skill as item costs are
zz_valyria_current_potion_cost = {
	value = 0
	if = {
		limit = { exists = var:zz_current_potion_cost }
		add = var:zz_current_potion_cost
		divide = { # the crafter's skill lowers the cost, to half at most (Higher Mysteries 100)
			value = zz_valyria_current_crafter_stat
			max = 100
			divide = 50
			min = 1
		}
	}
}
# The strength a crafter brews at: Alchemy skill / 10, from 1 to 10 (as zz_valyria_save_potion_strength_effect). Character scope
zz_valyria_potion_strength_value = {
	value = zz_valyria_magic_alchemy_magic_base_value
	divide = 10
	min = 1
	max = 10
}
# How many years a brewed draught (and Purging's disease immunity) lasts. Character scope (the brewer); stored on the potion as var:potion_duration
zz_valyria_potion_duration_value = {
	value = zz_valyria_potion_strength_value
}
# How many times a brewed Healing potion heals. Character scope (the brewer); stored on the potion as var:potion_heal
zz_valyria_potion_heal_value = {
	value = zz_valyria_potion_strength_value
}
# Read from the potion saved as scope:potion when it is drunk. Potions brewed before these were stored fall back to their strength
zz_valyria_drunk_potion_duration_value = {
	value = scope:potion.var:potion_strength
	if = {
		limit = { scope:potion = { has_variable = potion_duration } }
		value = scope:potion.var:potion_duration
	}
}
zz_valyria_drunk_potion_heal_value = {
	value = scope:potion.var:potion_strength
	if = {
		limit = { scope:potion = { has_variable = potion_heal } }
		value = scope:potion.var:potion_heal
	}
}
# Drops of power a Tonic of Power gives: 5 per point of potency
zz_valyria_drunk_potion_mana_value = {
	value = scope:potion.var:potion_strength
	multiply = 5
}
