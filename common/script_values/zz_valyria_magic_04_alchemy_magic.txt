### Alchemy Magic
zz_valyria_magic_alchemy_magic_base_value = { 
	scope:secondary_recipient = { 
		if = {
			limit = {
				exists = var:casttype
			}
			if = {
				limit = { 
					var:casttype ?= flag:potion 
				}
				value =  var:cast_potency
			}
			else_if = {
				limit = {
					var:casttype ?= flag:scroll
				}
				value =  var:cast_potency
			}
			else = { 
				value = "has_trait_xp(zz_magister|alchemy_magic)" 
				if = {
					limit = {
						has_variable = alchemy_magic_overcap 
					}
					add = this.var:alchemy_magic_overcap
				}
			}
		}
		else = { 
			value = "has_trait_xp(zz_magister|alchemy_magic)" 
			if = {
				limit = {
					has_variable = alchemy_magic_overcap 
				}
				add = this.var:alchemy_magic_overcap
			}
		}
		multiply = {
			value = 1
			if = {
				limit = {
					faith = { has_doctrine_parameter = zz_tenet_strength_alchemy_magic }
				}
				add = 0.25
			}
			if = {
				limit = {
					has_personal_tenet_flag = zz_personal_strength_alchemy_magic
				}
				add = 0.1
			}
		}
	}
}

# Alchemy age manipulation: the recipient's age as the counter widget shows it, using effective age for the immortal and long-lived
zz_valyria_alchemy_magic_recipient_age = {
	value = scope:recipient.age
	if = {
		limit = {
			scope:recipient = {
				OR = {
					has_trait = immortal
					has_trait = zz_valyrian_longevity
				}
			}
		}
		value = scope:recipient.effective_age
	}
}
