tenet_zz_blood_rites = {
	icon = tenet_gruesome_festivals

	piety_cost = {
		value = faith_tenet_cost_mid
		if = {
			limit = { rite_has_tenet = tenet_zz_blood_rites }
			multiply = faith_unchanged_doctrine_cost_mult
		}
	}

	can_pick = {
		OR = {
			NOT = { exists = scope:actor }
			scope:actor = { zz_valyria_is_mage = yes }
		}
	}

	can_pick_as_personal_tenet = {
		has_trait = zz_magister
	}

	personal_tenet_parameters = {
		zz_personal_strength_blood_magic
	}

	parameters = {
		zz_sorcery_accepted

	}
