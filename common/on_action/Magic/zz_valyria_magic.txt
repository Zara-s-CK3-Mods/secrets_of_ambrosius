# Soothsaying (Longevity): a vision turns aside a natural death, using one charge and adding a stacking boost to life expectancy and health
# Called from on_natural_death_second_chance in common/on_action/death.txt. Root is the character who would have died
zz_valyria_longevity_second_chance = {
	effect = {
		random_owned_story = {
			limit = {
				story_type = zz_valyria_magic_dream_magic_081_story
				var:mystery_duration > 0
			}
			change_variable = {
				name = mystery_duration
				subtract = 1
			}
			var:mystery_caster ?= {
				save_scope_as = caster
			}
		}
		add_character_modifier = {
			modifier = zz_valyria_magic_dream_magic_081_stacking_modifier
		}
		save_scope_as = zz_story_owner # the message icons need a saved scope
		scope:caster ?= {
			send_interface_message = {
				type = send_interface_message_as_feed_spell_buff_proc
				title = zz_valyria_magic_dream_magic.081.buff.toast.t
				desc = zz_valyria_magic_dream_magic.081.buff.toast.desc
				left_icon = scope:caster
				right_icon = scope:zz_story_owner
				custom_tooltip = zz_valyria_magic_dream_magic.081.buff.toast.tt
			}
		}
	}
}
