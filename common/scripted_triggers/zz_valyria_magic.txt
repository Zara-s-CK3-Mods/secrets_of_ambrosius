zz_valyria_has_mana_to_cast = { 
	OR = {
		has_character_modifier = zz_valyria_magic_drops_power
		$MANACOST$ <= this.zz_valyria_drops_of_power_stat
	}
}





zz_valyria_is_mage = {
	OR = { 
		has_trait = zz_magister
		any_spouse = {
			has_trait = zz_magister 
		}
		house ?= { 
			any_house_mage = {
				exists = this 
			}
		}
		any_court_mage = {
			exists = this 
		}
		any_realm_mage = {
			exists = this 
		}
	}
}



zz_valyria_can_become_apprentice = {
	OR = {
		NOT = { has_trait = zz_magister }
		
		
		any_spouse = {
			NOT = { has_trait = zz_magister }
		}
		any_close_or_extended_family_member = {
			NOT = { has_trait = zz_magister }
		}
		house ?= { 
			any_house_member = {
				NOT = { has_trait = zz_magister }
			}
		}
		any_courtier_or_guest = { 
			NOT = { has_trait = zz_magister }
		}
		any_vassal_or_below = { 
			NOT = { has_trait = zz_magister }
		}
	}
}

			has_trait = zz_magister

		}

			has_trait = zz_magister 
			NOT = { 
				has_character_flag = zz_valyria_magic_taught_magic 
			}
		}
		house ?= { 
			any_house_mage = {
				exists = this 
				NOT = { 
					has_character_flag = zz_valyria_magic_taught_magic 
				}
			}
		}
		any_court_mage = {
			exists = this 
			NOT = { 
				has_character_flag = zz_valyria_magic_taught_magic 
			}
		}
		any_realm_mage = {
			exists = this 
			NOT = { 
				has_character_flag = zz_valyria_magic_taught_magic 
			}
		}
	}
}


zz_valyria_knows_spell = {
	any_memory = {
		has_memory_type = $SPELL$
	}		
}

zz_valyria_knows_spell_name = {
	save_temporary_scope_as = spell_known
	scope:spell_learner = {
		any_memory = {
			var:spell_name ?= scope:spell_known
		}	
	}
		
}

zz_valyria_knows_spell_court = {
	OR = { 
		zz_valyria_knows_spell = {
			SPELL = $SPELL$
		}
		any_living_mage = {
			liege_or_court_owner ?= scope:actor 
			zz_valyria_knows_spell = {
				SPELL = $SPELL$
			}
		}
	}
}


zz_valyria_valid_caster = {
	zz_valyria_knows_spell = {
		SPELL = $SPELL$
	}
	zz_valyria_has_mana_to_cast = {
		MANACOST = $MANACOST$
	}
	is_incapable = no 
}

zz_valyria_valid_caster_court = {
	OR = { 
		zz_valyria_valid_caster = {
			SPELL = $SPELL$
			MANACOST = $MANACOST$
		}
		any_spouse = {
			has_trait = zz_magister 
			zz_valyria_valid_caster = {
				SPELL = $SPELL$
				MANACOST = $MANACOST$
			}
		}
		house ?= { 
			any_house_mage = {
				zz_valyria_valid_caster = {
					SPELL = $SPELL$
					MANACOST = $MANACOST$
				}
			}
		}
		any_court_mage = {
			zz_valyria_valid_caster = {
				SPELL = $SPELL$
				MANACOST = $MANACOST$
			}
		}
		any_realm_mage = {
			zz_valyria_valid_caster = {
				SPELL = $SPELL$
				MANACOST = $MANACOST$
			}
		}
	}
}
