# Spell: Air 001 - Mystery of the Slicing Wind
zz_valyria_magic_air_magic_001_effect = {
	random_list = {
		75 = {
			modifier = {
				add = zz_valyria_magic_$SPELLSCHOOL$_$SPELL$_value
			}
			zz_valyria_magic_spell_success_interface = {
				SPELLSCHOOL = $SPELLSCHOOL$
				SPELL = $SPELL$
			}
			zz_valyria_magic_duel_stress_effect = yes
			scope:recipient = {
				hidden_effect = {
					force_character_skill_recalculation = yes
					add_character_modifier = duel_edge_debuff_02_modifier
				}
			}
			# The first event option spares the target; otherwise losing the combat can be fatal
			if = {
				limit = { scope:option_01 ?= yes }
				zz_configure_start_magic_combat_effect = {
					SC_INITIATOR = scope:secondary_recipient
					SC_ATTACKER = scope:secondary_recipient
					SC_DEFENDER = scope:recipient
					FATALITY = no
					FIXED = no
					LOCALE = tavern
					OUTPUT_EVENT = jason_locale_events.201
					INVALIDATION_EVENT = single_combat.1006
					MAGIC = air
				}
			}
			else = {
				zz_configure_start_magic_combat_effect = {
					SC_INITIATOR = scope:secondary_recipient
					SC_ATTACKER = scope:secondary_recipient
					SC_DEFENDER = scope:recipient
					FATALITY = possible
					FIXED = no
					LOCALE = tavern
					OUTPUT_EVENT = jason_locale_events.201
					INVALIDATION_EVENT = single_combat.1006
					MAGIC = air
				}
			}
		}
		25 = {
			zz_valyria_magic_spell_failure_interface = {
				SPELLSCHOOL = $SPELLSCHOOL$
				SPELL = $SPELL$
			}
		}
	}
}
