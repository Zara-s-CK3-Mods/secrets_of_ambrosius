# Creates a potion for the scoped character. EFFECT is the potion key; run zz_valyria_save_potion_strength_effect first
# The potion keeps its own potency, duration and heal strength, fixed when it was brewed
zz_valyria_create_potion_effect = {
	set_artifact_rarity_masterwork = yes
	create_artifact = {
		name = zz_valyria_potion_$EFFECT$_name
		description = zz_valyria_potion_$EFFECT$_desc
		type = potion
		visuals = flask
		wealth = scope:wealth
		quality = scope:quality
		modifier = artifact_placeholder_modifier
		save_scope_as = newly_created_potion
		decaying = no
	}
	scope:newly_created_potion ?= {
		clear_artifact_modifiers = yes
		set_variable = { name = potion_effect value = flag:$EFFECT$ }
		set_variable = { name = potion_strength value = scope:potion_strength }
		set_variable = { name = potion_duration value = scope:potion_duration }
		set_variable = { name = potion_heal value = scope:potion_heal }
	}
}

# Saves the brewer's potency, duration (years) and heal strength as scope values for zz_valyria_create_potion_effect. Run in the brewer's scope
zz_valyria_save_potion_strength_effect = {
	save_scope_value_as = { name = potion_strength value = zz_valyria_potion_strength_value }
	save_scope_value_as = { name = potion_duration value = zz_valyria_potion_duration_value }
	save_scope_value_as = { name = potion_heal value = zz_valyria_potion_heal_value }
}

# Applies the potion saved as scope:potion to the scoped character, then the potion is used up
zz_valyria_drink_potion_effect = {
	switch = {
		trigger = scope:potion.var:potion_effect
		flag:elixir_longevity = { add_character_modifier = { modifier = zz_valyria_potion_elixir_longevity_modifier } }
		flag:elixir_fertility = { add_character_modifier = { modifier = zz_valyria_potion_elixir_fertility_modifier } }
		flag:draught_strength = {
			add_character_modifier = {
				modifier = zz_valyria_potion_draught_strength_modifier
				years = zz_valyria_drunk_potion_duration_value
			}
		}
		flag:draught_clarity = {
			add_character_modifier = {
				modifier = zz_valyria_potion_draught_clarity_modifier
				years = zz_valyria_drunk_potion_duration_value
			}
		}
		flag:draught_calm = {
			add_character_modifier = {
				modifier = zz_valyria_potion_draught_calm_modifier
				years = zz_valyria_drunk_potion_duration_value
			}
		}
		flag:draught_warding = {
			add_character_modifier = {
				modifier = zz_valyria_potion_draught_warding_modifier
				years = zz_valyria_drunk_potion_duration_value
			}
		}
		flag:potion_healing = {
			while = {
				count = zz_valyria_drunk_potion_heal_value
				zz_valyria_magic_dream_magic_058_heal_effect = yes
			}
		}
		flag:potion_purging = {
			add_character_flag = {
				flag = immune_to_disease
				years = zz_valyria_drunk_potion_duration_value
			}
			if = {
				limit = { has_any_disease_trigger = yes }
				add_character_modifier = {
					modifier = zz_valyria_potion_purging_treatment_modifier
					years = 1
				}
			}
		}
		flag:potion_restoring = {
			if = {
				limit = { has_variable = zz_miscast_count }
				change_variable = {
					name = zz_miscast_count
					subtract = scope:potion.var:potion_strength
				}
				if = {
					limit = { var:zz_miscast_count < 0 }
					set_variable = { name = zz_miscast_count value = 0 }
				}
			}
		}
		flag:tonic_power = {
			zz_valyria_mana_gain = { MANAGAIN = zz_valyria_drunk_potion_mana_value } # $MANAGAIN$ is pasted as text, so it takes a named value, not a block
		}
	}
	destroy_artifact = scope:potion
}
