# Effects behind the Magical Archives character pickers. Each expects scope:zz_magic_user (the player) and the chosen character
# under the picker's scope name. Used by the archive scripted GUIs and zz_valyria_archive_select_character_interaction

# zz_valyria_mages_selection: scope:zz_magic_caster
zz_valyria_archive_set_mage_effect = {
	scope:zz_magic_user = {
		set_variable = {
			name = zz_magic_caster
			value = scope:zz_magic_caster
		}
	}
}

# zz_valyria_magic_scroll_maker: scope:zz_scroll_maker
zz_valyria_archive_set_scroll_maker_effect = {
	scope:zz_magic_user = {
		clear_variable_list = zz_valyria_spell_list
		set_variable = {
			name = zz_scroll_maker
			value = scope:zz_scroll_maker
		}
	}
	scope:zz_scroll_maker = {
		every_memory = {
			limit = {
				has_memory_category = magic
			}
			scope:zz_magic_user = {
				add_to_variable_list = {
					name = zz_valyria_spell_list
					target = prev
				}
			}
		}
	}
}
