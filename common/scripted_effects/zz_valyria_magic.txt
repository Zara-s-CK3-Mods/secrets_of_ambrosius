zz_mage_contract_travel_or_progress_effect = {
	if = {
		limit = { location != $DESTINATION$ }
		start_travel_plan = {
			destination = $DESTINATION$
			on_arrival_event = $ARRIVAL_EVENT$
			on_travel_planner_cancel_event = zz_valyria_magic_contracts.0001
			on_arrival_destinations = all_but_last
		}
	}
	else = { trigger_event = $ARRIVAL_EVENT$ }
}

populate_location_with_magic_contracts_effect = {
	if = {
		limit = {
			# The AI have their own system for getting contracts.
			trigger_if = {
				zz_valyria_is_mage = yes 
				is_ai = no
			}
			lock_contracts_from_spawning_in_sahara_trigger = yes
		}
		# Flag our pre-existing contracts if necessary.
		if = {
			limit = { exists = scope:toggle_distance_ignore }
			every_character_task_contract = { add_to_list = pre_existing_contracts_list }
		}
		# Now populate.
		hidden_effect = { # internal setup; the contracts themselves are announced
			populate_task_contracts_for_area = {
				location = $AREA_CHAR$.capital_province
				amount = $AMOUNT$
				group = { 
					zz_valyria_magic_contracts
				}
			}
		}
		# Then flag our new contracts (again, if necessary).
		if = {
			limit = { exists = scope:toggle_distance_ignore }
			every_character_task_contract = {
				limit = {
					NOT = { is_in_list = pre_existing_contracts_list }
				}
				set_variable = {
					name = retain_despite_distance
					value = yes
				}
			}
		}
	}
}

magic_contract_passive_spawn_effect = {
	#count how many contracts you already have
	every_character_task_contract = {
		limit = {
			has_task_contract_group = zz_valyria_magic_contracts
		}
		add_to_list = num_of_magic_contracts_before
	}
	# G'is a contract.
	populate_location_with_magic_contracts_effect = {
		AREA_CHAR = root
		AMOUNT = $SPAWN_CONTRACTS$
	}
	
	#count them again to see if the effect above spawned something
	every_character_task_contract = {
		limit = {
			has_task_contract_group = zz_valyria_magic_contracts
		}
		add_to_list = num_of_magic_contracts_after
	}
	set_local_variable = {
		name = current_value
		value = list_size:num_of_magic_contracts_before
	}
	# Make sure we don't get too many of them just sitting in one place.
	while = {
		limit = {
			local_var:current_value < list_size:num_of_magic_contracts_after
		}
		increment_variable_effect = {
			VAR = magic_contract_passive_spawn_tally
			VAL = 1
		}
		change_local_variable = {
			name = current_value
			add = 1
		}
	}
}

zz_valyria_magic_prophecy_spawn = {
	if = {
		limit = { exists = scope:toggle_distance_ignore }
		every_character_task_contract = { add_to_list = pre_existing_contracts_list }
	}
	# Now populate.
	random_ruler = {
		save_scope_as = ruler_location
	}
	create_task_contract = {
		task_contract_type = $PROPHECY$
		location = scope:ruler_location.capital_province
		task_contract_employer = scope:ruler_location
		task_contract_tier = 4
		save_scope_as = new_contract
		target = scope:secondary_recipient 
	}
	scope:new_contract = {
		set_variable = {
			name = zz_prophet
			value = scope:secondary_recipient
		}
	}
	# Then flag our new contracts (again, if necessary).
	if = {
		limit = { exists = scope:toggle_distance_ignore }
		every_character_task_contract = {
			limit = {
				NOT = { is_in_list = pre_existing_contracts_list }
			}
			set_variable = {
				name = retain_despite_distance
				value = yes
			}
		}
	}
}

# As zz_valyria_magic_prophecy_spawn, but the prophecy is set at a chosen province, employed by that county's holder
zz_valyria_magic_prophecy_location_spawn = {
	if = {
		limit = { exists = scope:toggle_distance_ignore }
		every_character_task_contract = { add_to_list = pre_existing_contracts_list }
	}
	$LOCATION$ = {
		save_scope_as = prophecy_location
		county.holder ?= { save_scope_as = ruler_location }
	}
	if = {
		limit = { NOT = { exists = scope:ruler_location } }
		random_ruler = {
			save_scope_as = ruler_location
		}
	}
	create_task_contract = {
		task_contract_type = $PROPHECY$
		location = scope:prophecy_location
		task_contract_employer = scope:ruler_location
		task_contract_tier = 4
		save_scope_as = new_contract
		target = scope:secondary_recipient
	}
	scope:new_contract = {
		set_variable = {
			name = zz_prophet
			value = scope:secondary_recipient
		}
	}
	if = {
		limit = { exists = scope:toggle_distance_ignore }
		every_character_task_contract = {
			limit = {
				NOT = { is_in_list = pre_existing_contracts_list }
			}
			set_variable = {
				name = retain_despite_distance
				value = yes
			}
		}
	}
}
