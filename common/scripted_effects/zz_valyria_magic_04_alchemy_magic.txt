		75 = {
			modifier = {
				add = zz_valyria_magic_$SPELLSCHOOL$_$SPELL$_value

			zz_valyria_magic_spell_success_interface = {

					callous = minor_stress_gain
				}
			}
			# Change age by the amount chosen on the counter widget
			if = {
				limit = {
					scope:recipient = {
						OR = {
							has_trait = immortal

						}

				}
				set_variable = {
					name = immortal_effective_age
					value = scope:recipient.effective_age
					days = 1

					name = immortal_effective_age

				}
				scope:recipient = {
					set_immortal_age = root.var:immortal_effective_age
				}
			}
			else = {
				scope:recipient = {
					change_age = zz_valyria_magic_$SPELLSCHOOL$_$SPELL$_value_03
				}
			}
			scope:recipient = {
				if = {
					limit = {
						NOT = {
							scope:actor = scope:recipient
						}
					}
					add_opinion = {
						modifier = grateful_opinion

					}
				}
			}
		}
		25 = {
			zz_valyria_magic_spell_failure_interface = {
