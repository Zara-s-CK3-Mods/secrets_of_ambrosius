# Spell: Water 001 - Mystery of the Crushing Waters
zz_valyria_magic_water_magic_001_effect = {
	random_list = {
		75 = {
			modifier = {
				add = zz_valyria_magic_$SPELLSCHOOL$_$SPELL$_value
			}
			zz_valyria_magic_spell_success_interface = {
				SPELLSCHOOL = $SPELLSCHOOL$
				SPELL = $SPELL$
			}
			zz_valyria_magic_duel_stress_effect = yes
			scope:recipient = {
				hidden_effect = {
					force_character_skill_recalculation = yes
					add_character_modifier = duel_edge_debuff_02_modifier
				}
			}
			# The first event option spares the target; otherwise losing the combat can be fatal
			if = {
				limit = { scope:option_01 ?= yes }
				zz_configure_start_magic_combat_effect = {
					SC_INITIATOR = scope:secondary_recipient
					SC_ATTACKER = scope:secondary_recipient
					SC_DEFENDER = scope:recipient
					FATALITY = no
					FIXED = no
					LOCALE = tavern
					OUTPUT_EVENT = jason_locale_events.201
					INVALIDATION_EVENT = single_combat.1006
					MAGIC = water
				}
			}
			else = {
				zz_configure_start_magic_combat_effect = {
					SC_INITIATOR = scope:secondary_recipient
					SC_ATTACKER = scope:secondary_recipient
					SC_DEFENDER = scope:recipient
					FATALITY = possible
					FIXED = no
					LOCALE = tavern
					OUTPUT_EVENT = jason_locale_events.201
					INVALIDATION_EVENT = single_combat.1006
					MAGIC = water
				}
			}
		}
		25 = {
			zz_valyria_magic_spell_failure_interface = {
				SPELLSCHOOL = $SPELLSCHOOL$
				SPELL = $SPELL$
			}
		}
	}
}

							trigger = has_trait

							}

							}
							early_great_pox = {
								remove_trait = early_great_pox
							}
							lovers_pox = {
								remove_trait = lovers_pox
							}
							leper = {
								remove_trait = leper
							}
							gout_ridden = {
								recover_from_disease_effect = {
									DISEASE = gout_ridden
								}								#Removes the trait, sends event "health.2202" to those who care
								remove_disease_treatment_effect = yes
							}
							consumption = {
								recover_from_disease_effect = {
									DISEASE = consumption
								}								#Removes the trait, sends event "health.2202" to those who care
								remove_disease_treatment_effect = yes
							}
							cancer = {
								recover_from_disease_effect = {
									DISEASE = cancer
								}								#Removes the trait, sends event "health.2202" to those who care
								remove_disease_treatment_effect = yes
							}
							typhus = {
								recover_from_disease_effect = {
									DISEASE = typhus
								}								#Removes the trait, sends event "health.2202" to those who care
								remove_disease_treatment_effect = yes
							}
							bubonic_plague = {
								recover_from_disease_effect = {
									DISEASE = bubonic_plague
								}								#Removes the trait, sends event "health.2202" to those who care & liege/host/imprisoner, adds immunity flag
								remove_disease_treatment_effect = yes
							}
							smallpox = {
								recover_from_disease_effect = {

							}

								remove_disease_treatment_effect = yes
							}
							dysentery = {
