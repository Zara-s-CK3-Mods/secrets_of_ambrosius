# Spell: Dream 102 - Mystery of Secret Weaving (False Memory)
# The subject wakes remembering something they did to another character, and a secret is made of it.
# Unlike the Shadow of Guilt, the secret involves a second, chosen character. The commander learns it.
zz_valyria_magic_dream_magic_102_effect = {
	random_list = {
		75 = {
			modifier = {
				add = zz_valyria_magic_$SPELLSCHOOL$_$SPELL$_value
			}
			zz_valyria_magic_spell_success_interface = { SPELLSCHOOL = $SPELLSCHOOL$ SPELL = $SPELL$}
			scope:recipient = {
				switch = {
					trigger = scope:option
					# A love affair: the dream makes them lovers, so the secret holds. Not possible with a spouse or betrothed
					flag:zz_weave_lover = {
						if = {
							limit = {
								is_adult = yes
								scope:target = { is_adult = yes is_alive = yes }
								NOR = {
									is_consort_of = scope:target
									betrothed ?= scope:target
								}
							}
							if = {
								limit = { NOT = { has_relation_lover = scope:target } }
								set_relation_lover = scope:target
							}
							if = {
								limit = {
									NOT = { any_secret = { secret_type = secret_lover secret_target = scope:target } }
								}
								add_secret = { type = secret_lover target = scope:target }
							}
							random_secret = {
								limit = { secret_type = secret_lover secret_target = scope:target }
								reveal_to = scope:actor
							}
						}
					}
					# An attempt on the chosen character's life
					flag:zz_weave_murder_attempt = {
						if = {
							limit = {
								NOT = { any_secret = { secret_type = secret_murder_attempt secret_target = scope:target } }
							}
							add_secret = { type = secret_murder_attempt target = scope:target }
						}
						random_secret = {
							limit = { secret_type = secret_murder_attempt secret_target = scope:target }
							reveal_to = scope:actor
						}
					}
				}
			}
		}
		25 = {
			zz_valyria_magic_spell_failure_interface = { SPELLSCHOOL = $SPELLSCHOOL$ SPELL = $SPELL$}
		}
	}
}
