zz_valyria_army_cast_click = {
	scope = character
	is_shown = {
		zz_open_mage_window = yes
	}
	is_valid = {

	}
	effect = {
		save_scope_as = actor
		scope:actor = {
			if = {
				limit = {
					NOT = { exists = var:zz_miscast_count }
				}
				set_variable = {
					name = zz_miscast_count
					value = 0
				}
			}
		}
		remove_variable = zz_selected_mystery_province
		remove_variable = zz_selected_mystery_county
		# Armies at sea have no county, so province spells fall back to the recipient's lands
		if = {
			limit = {
				exists = scope:selected_province.county
			}
			set_variable = {
				name = zz_selected_mystery_province
				value = scope:selected_province
			}
			set_variable = {
				name = zz_selected_mystery_county
				value = scope:selected_province.county
			}
		}
		# The combat window has no army to pass, so use this side's largest army in the battle
		if = {
			limit = {
				NOT = { exists = scope:selected_army }
			}
			scope:opened_character = {
				ordered_army = {
					limit = {
						location = scope:selected_province
					}
					order_by = army_size
					save_scope_as = selected_army
				}
			}
		}
		scope:opened_character = {
			save_scope_as = recipient
		}
		zz_ui_setup_effect = yes
		if = {
			limit = {
				exists = scope:selected_army
			}
			set_variable = {
				name = zz_selected_mystery_army
				value = scope:selected_army
			}
		}
	}
}

zz_valyria_magic_cast = {
	scope = character
	is_valid = {
		exists = var:zz_valyria_mystery
		var:zz_magic_caster = {
			zz_valyria_has_mana_to_cast = {
				MANACOST = root.var:zz_valyria_mystery.mystery_cost
			}
		}
		
	}
	effect = {
		save_scope_as = actor
		var:zz_magic_caster = {
			save_scope_as = secondary_recipient
		}
		var:zz_magic_target = { 
			save_scope_as = recipient
		}
		var:zz_valyria_mystery = {
			save_scope_as = zz_valyria_mystery
		}
		scope:secondary_recipient = { 
			set_variable = {
				name = zz_manacost 
				value = current_mystery_cost
			}
			zz_valyria_mana_spend = yes
		}
		root = {
			#remove_variable = zz_magic_ui
		}
		scope:zz_valyria_mystery.var:spell_school = {
			save_scope_as = spell_school
		}
		scope:zz_valyria_mystery.var:spell_number = {
			save_scope_as = spell_number
		}
		scope:zz_valyria_mystery.var:spell_name = {
			save_scope_as = spell_name
		}
		zz_spell_trigger_switch_effect = { SCOPE = scope:zz_valyria_mystery.var:spell_name  EFFECT = zz_valyria_mystery_casting_gui_effect }
	}
}
