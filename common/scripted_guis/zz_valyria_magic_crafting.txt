zz_valyria_magic_item_selection = {
	scope = character
	saved_scopes = {
		zz_selected_item
	}
	is_shown = {
		var:zz_selected_item ?= scope:zz_selected_item
	}
	effect = {
		root = {
			set_variable = {
				name = zz_selected_item
				value = scope:zz_selected_item
			}
			zz_valyria_crafting_calculate_crafting_cost_item_effect = yes 
			debug_log_scopes = yes
			debug_log = "Item Selected"
			debug_log = debug_var

			if = { 
				limit = {
					scope:zz_selected_item = flag:item_sword
				}
				remove_variable = zz_sword_selection_sword
				set_variable = {
					name = zz_item_ui
					value = flag:sword_crafting_ui
				}
			}
			else_if = {
				limit = {
					scope:zz_selected_item = flag:item_ring
				}
				remove_variable = zz_ring_selection_ring
				set_variable = {
					name = zz_item_ui
					value = flag:ring_crafting_ui
				}
			}
		}
	}
}
# Enchantment slots 1 to 4. A slot can't take an enchantment already in another slot; picking closes the popup
zz_valyria_magic_modifier_selection = {
	scope = character
	saved_scopes = {
		zz_selected_modifier
	}
	is_shown = {
		var:zz_selected_modifier ?= scope:zz_selected_modifier
	}
	is_valid = {
		OR = {
			scope:zz_selected_modifier = flag:item_modifier_none
			NOR = {
			var:zz_selected_modifier_2 ?= scope:zz_selected_modifier
			var:zz_selected_modifier_3 ?= scope:zz_selected_modifier
			var:zz_selected_modifier_4 ?= scope:zz_selected_modifier
			}
		}
	}
	effect = {
		root = {
			set_variable = {
				name = zz_selected_modifier
				value = scope:zz_selected_modifier
			}
			zz_valyria_crafting_calculate_crafting_cost_modifier_effect = yes
			remove_variable = zz_enchant_popup
		}
	}
}

# Potions tab: brew the chosen potion for yourself. Uses crafting's success chance; the cost is paid either way, as for items
zz_valyria_magic_potion_brew = {
	scope = character
	is_valid = {
		custom_tooltip = {
			text = zz_valyria_potion_choose_first_tt
			exists = var:zz_selected_potion
		}
		exists = var:zz_item_crafter
		var:zz_selected_potion = { save_temporary_scope_as = zz_selected_potion }
		zz_valyria_crafter_can_brew_potion_trigger = yes
		zz_valyria_resources_current_spend >= zz_valyria_current_potion_cost
	}
	effect = {
		save_scope_as = recipient
		save_scope_as = actor
		var:zz_item_crafter = {
			save_scope_as = zz_item_crafter
			save_scope_as = secondary_recipient
			zz_valyria_save_potion_strength_effect = yes
		}
		remove_treasury_or_gold = zz_valyria_current_potion_cost
		random_list = {
			65 = {
				modifier = {
					add = zz_spellcast_value
				}
				switch = {
					trigger = var:zz_selected_potion
						flag:elixir_longevity = { zz_valyria_create_potion_effect = { EFFECT = elixir_longevity } }
						flag:elixir_fertility = { zz_valyria_create_potion_effect = { EFFECT = elixir_fertility } }
						flag:draught_strength = { zz_valyria_create_potion_effect = { EFFECT = draught_strength } }
						flag:draught_clarity = { zz_valyria_create_potion_effect = { EFFECT = draught_clarity } }
						flag:draught_calm = { zz_valyria_create_potion_effect = { EFFECT = draught_calm } }
						flag:draught_warding = { zz_valyria_create_potion_effect = { EFFECT = draught_warding } }
						flag:potion_healing = { zz_valyria_create_potion_effect = { EFFECT = potion_healing } }
						flag:potion_purging = { zz_valyria_create_potion_effect = { EFFECT = potion_purging } }
						flag:potion_restoring = { zz_valyria_create_potion_effect = { EFFECT = potion_restoring } }
						flag:tonic_power = { zz_valyria_create_potion_effect = { EFFECT = tonic_power } }
				}
				send_interface_toast = {
					type = event_toast_effect_good
					title = zz_valyria_potion_brewed_toast
					left_icon = scope:zz_item_crafter
				}
			}
			35 = {
				send_interface_toast = {
					type = event_toast_effect_bad
					title = zz_valyria_potion_spoiled_toast
					left_icon = scope:zz_item_crafter
				}
			}
		}
	}
}

# Ring look picker: var:zz_ring_look holds the chosen look (flag:ring_look_<school>); flag:ring_look_default clears it,
# so the ring wears its own school's look. The popup is open while var:zz_ring_look_popup is set
zz_ring_look_popup_open = {
	scope = character
	effect = { set_variable = { name = zz_ring_look_popup value = yes } }
}
zz_ring_look_popup_close = {
	scope = character
	effect = { remove_variable = zz_ring_look_popup }
}
zz_ring_look_selection = {
	scope = character
	saved_scopes = {
		zz_ring_look
	}
	is_shown = {
		OR = {
			var:zz_ring_look ?= scope:zz_ring_look
			AND = {
				scope:zz_ring_look = flag:ring_look_default
				NOT = { has_variable = zz_ring_look }
			}
		}
	}
	effect = {
		if = {
			limit = { scope:zz_ring_look = flag:ring_look_default }
			remove_variable = zz_ring_look
		}
		else = {
			set_variable = { name = zz_ring_look value = scope:zz_ring_look }
		}
		remove_variable = zz_ring_look_popup
	}
}
