				}
				if = {
					limit = { sub_realm_size > 1 }
					add = {
						value = @republic_type_realm_size_popular_prestige_cost
						multiply = sub_realm_size # Per county within the (sub)realm
						desc = realm_size
					}
				}
				# Popular opinion - add/subtract 0.5% of base cost per opinion
				if = {
					limit = {
						capital_county ?= { popular_opinion > 0 }
					}
					subtract = {
						value = @republic_type_prestige_cost
						multiply = {
							value = capital_county.popular_opinion
							divide = 200
						}
						desc = popular_opinion_concept_desc
					}
				}
				else_if = {
					limit = {
						capital_county ?= { popular_opinion < 0 }
					}
					add = {
						value = @republic_type_prestige_cost
						multiply = {
							value = capital_county.popular_opinion
							divide = 200
						}
						desc = popular_opinion_concept_desc
					}
				}
				# Title tier
				if = {
					limit = {
						highest_held_title_tier >= tier_kingdom
					}
					add = {
						value = @republic_type_prestige_cost
						switch = {
							trigger = primary_title.tier
							tier_hegemony = { multiply = 7 }
							tier_empire = { multiply = 3 }
						}
						desc = title_tier_desc
					}
				}
				else_if = {
					limit = {
						highest_held_title_tier = tier_county
					}
					subtract = {
						value = @republic_type_prestige_cost
						divide = 2
						desc = title_tier_desc
					}
				}
			}
			treasury = {
				add = {
					value = @republic_type_treasury_cost
					desc = BASE
				}
				if = {
					limit = { sub_realm_size > 1 }
					add = {
						value = @republic_type_realm_size_popular_treasury_cost
						multiply = sub_realm_size # Per county within the (sub)realm
						desc = realm_size
					}
				}
				# Popular opinion - add/subtract 0.5% of base cost per opinion
				if = {
					limit = {
						capital_county ?= { popular_opinion > 0 }
					}
					subtract = {
						value = @republic_type_treasury_cost
						multiply = {
							value = capital_county.popular_opinion
							divide = 200
						}
						desc = popular_opinion_concept_desc
					}
				}
				else_if = {
					limit = {
						capital_county ?= { popular_opinion < 0 }
					}
					add = {
						value = @republic_type_treasury_cost
						multiply = {
							value = capital_county.popular_opinion
							divide = 200
						}
						desc = popular_opinion_concept_desc
